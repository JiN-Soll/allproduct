} // ── 로그인 → 쿠키 반환 ────────────────── function login() { try { // 1단계: 로그인 폼 GET (토큰/세션 획득) const loginPageRes = UrlFetchApp.fetch(SHOP_URL + '/member/login.html', { method: 'GET', followRedirects: true, muteHttpExceptions: true }); '); } function mergeCookies(c1, c2) { if (!c1) return c2 || ''; if (!c2) return c1 || ''; const map = {}; (c1 + '; ' + c2).split(';').forEach(part => { const kv = part.trim(); if (!kv) return; const eqIdx = kv.indexOf('=');
